from datetime import timedelta
from django.db.models import Q, F, Count
from django.utils import timezone
from webapp.models import Task, TaskType


Task.objects.filter(status__name='Done', updated_at__gte=timezone.now() - timedelta(days=30))


Task.objects.filter(status__name__in=['New', 'In Progress'],
                    types__name__in=['Bug', 'Enhancement']).distinct()


Task.objects.filter(Q(summary__icontains='bug') | Q(types__name='Bug')).exclude(status__name='Done').distinct()


Task.objects.values('id', 'summary', 'types__name', 'status__name')


Task.objects.filter(summary=F('description'))


TaskType.objects.annotate(tasks_count=Count('tasks')).values('name', 'tasks_count')
